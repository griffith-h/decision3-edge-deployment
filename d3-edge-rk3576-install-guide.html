1. 准备自己的 RK3576 系统

基准在同一进程、4 个大核及 4 个 Torch 线程下，分别运行优化 CPU 和 RKNN。每个用例 / 后端先预热一次，再计时三次取中位数，覆盖 5 个用例、12 个问题，输入在附录的固定样本 JSON 中。

字段 / 指标验收含义
顶层 passed必须为 true；执行失败或 false 均不能视为通过
cases.*.npu.choices_unchanged分类选项必须与同轮优化 CPU 一致
max_probability_or_score_delta同名概率 / score 取最大绝对差，排除派生 confidence、legend；要求 ≤0.01
speedupCPU 三次中位数 ÷ RKNN 三次中位数；不包含 HTTP、模型加载和首次缓存建立
cpu_fallback_calls记录回退次数；大输入回退 CPU 属于预期边界

注意输出差异针对优化 CPU 基线，不是原始 BF16 CPU。模型内提速不能直接当作 HTTP 提速；若要测 HTTP，需要对相同输入、预热和负载条件下的 CPU / RKNN 接口分别计时。新机结果不必与历史数字逐位相同，但选项和差值必须通过验收。

9. 常见故障

现象检查与处理
缺少 /dev/dri 或 rknn_* 报错核对 RK3576、64 位系统、板厂 NPU 驱动；基线为驱动 0.9.8 + Runtime 2.5.0。先通过真实算子测试。容器不会补齐主机内核驱动。
Docker 权限错误使用本文 sudo docker 命令，确认 Docker 服务已启动。
镜像 / wheel / SDK 下载失败查看失败域名，检查 DNS / 代理和磁盘后重跑构建。不要任意替换为最新版依赖或跳过哈希。
模型下载失败修复网络后重跑独立下载步骤，保留 data 目录。可自行配置可信 HF_ENDPOINT，但应保持相同模型版本和文件哈希校验。
端口占用或容器重名改用空闲宿主端口或容器名；-p 的右侧仍为容器 8900。同步修改调用和维护命令。
健康接口未就绪检查 docker logs、磁盘和内存；等模型加载 / NPU 缓存完成。容器 running 不等于模型 ready。
启动后退出或频繁重启检查 OOM、模型哈希不符、驱动报错。先 docker stop，再排查；保留日志。
速度与历史差距大核对大核 4–7、温度降频、后台负载和输入长度。超过 512 矩阵行会回退 CPU；不要拿矩阵微基准倍数代替完整请求倍数。
选项不一致或概率差超标视为验证失败，保留 benchmark-all.json；核对模型版本、依赖及 SDK / 驱动组合后再分析。

    'librknnrt.so': 'rknpu2/runtime/Linux/librknn_api/aarch64/librknnrt.so',
    'rknn_api.h': 'rknpu2/runtime/Linux/librknn_api/include/rknn_api.h',

    'LICENSE': 'LICENSE',
}
destination.mkdir(parents=True, exist_ok=True)
for name, path in paths.items():
    for attempt in range(3):
        try:
            with urllib.request.urlopen(base + path, timeout=120) as response:
                data = response.read()
            break
        except OSError:
            if attempt == 2:
                raise
            time.sleep(2)
    if name in manifest['files']:
        assert hashlib.sha256(data).hexdigest() == manifest['files'][name]['sha256'], name
    (destination / name).write_bytes(data)
    print(f'Verified/downloaded official SDK: {name}', flush=True)

// Experimental, synchronous, single-owner handle. Callers must serialize calls.
static thread_local std::string last_error;
struct Matrix {
  rknn_matmul_ctx ctx = 0;
  rknn_tensor_mem *a = nullptr, *b = nullptr, *c = nullptr;
  int k, n, selected = -1;
  std::vector<rknn_matmul_shape> shapes;
  std::vector<rknn_matmul_io_attr> attrs;
  ~Matrix() {
    if (a) rknn_destroy_mem(ctx, a);
    if (b) rknn_destroy_mem(ctx, b);
    if (c) rknn_destroy_mem(ctx, c);
    if (ctx) rknn_matmul_destroy(ctx);
  }
};
static bool check(int code, const char* operation) {
  if (code == 0) return true;
  last_error = std::string(operation) + " returned " + std::to_string(code);
  return false;
}
extern "C" const char* mm_error() { return last_error.c_str(); }
extern "C" void* mm_create(int k, int n, const void* half_kn) {
  last_error.clear();
  if (k <= 0 || k > 10240 || k % 16 || n <= 0 || n % 8 || !half_kn) {
    last_error = "Unsupported K/N alignment or missing weights"; return nullptr;
  }
  auto p = std::make_unique<Matrix>();
  p->k = k; p->n = n;
  for (int m : {128, 256, 512}) p->shapes.push_back({m,k,n});
  p->attrs.resize(p->shapes.size());
  rknn_matmul_info info{};
  info.M=512; info.K=k; info.N=n;
  info.type=RKNN_FLOAT16_MM_FLOAT16_TO_FLOAT32;

  info.AC_layout=RKNN_MM_LAYOUT_NORM;

      p->shapes.size(), p->shapes.data(), p->attrs.data()), "create_dynamic")) return nullptr;

  if (!check(rknn_matmul_set_core_mask(p->ctx, RKNN_NPU_CORE_AUTO), "set_cores")) return nullptr;
  uint32_t as=0, bs=0, cs=0;
  for (auto& attr : p->attrs) {
    as=std::max(as,attr.A.size); bs=std::max(bs,attr.B.size); cs=std::max(cs,attr.C.size);
  }
  p->a=rknn_create_mem(p->ctx,as); p->b=rknn_create_mem(p->ctx,bs); p->c=rknn_create_mem(p->ctx,cs);

  std::memset(p->a->virt_addr,0,as);
  return p.release();
}

    if (!check(rknn_matmul_set_io_mem(p->ctx,p->a,&p->attrs[index].A),"bind_A")) return -1;
    if (!check(rknn_matmul_set_io_mem(p->ctx,p->c,&p->attrs[index].C),"bind_C")) return -1;
    p->selected=index;
  }
  size_t bytes=size_t(rows)*p->k*2;
  std::memcpy(p->a->virt_addr,half_mk,bytes);
  std::memset(static_cast<char*>(p->a->virt_addr)+bytes,0,size_t(p->shapes[index].M-rows)*p->k*2);
  if (!check(rknn_matmul_run(p->ctx),"run")) return -1;
  std::memcpy(out_mn,p->c->virt_addr,size_t(rows)*p->n*sizeof(float));
  return 0;
}

"""Experimental FP16 RKNN Linear adapter; original attention stays in PyTorch."""
import ctypes
import time
from pathlib import Path
from types import MethodType
import torch


class Engine:
    def __init__(self):
        self.lib = ctypes.CDLL(str(Path(__file__).with_name('libmatmul_bridge.so')))
        self.lib.mm_error.restype = ctypes.c_char_p
        self.lib.mm_create.argtypes = [ctypes.c_int, ctypes.c_int, ctypes.c_void_p]
        self.lib.mm_create.restype = ctypes.c_void_p
        self.lib.mm_run.argtypes = [ctypes.c_void_p, ctypes.c_int, ctypes.c_void_p, ctypes.c_void_p]
        self.lib.mm_run.restype = ctypes.c_int
        self.lib.mm_destroy.argtypes = [ctypes.c_void_p]
        self.handles = []
        self.calls = self.cpu_fallbacks = 0
        self.seconds = self.create_seconds = 0.0

    def error(self):
        return RuntimeError(self.lib.mm_error().decode())

    def create(self, weight):
        # PyTorch weight [N,K] -> official normal B layout [K,N], then native-pack once.
        start = time.perf_counter()
        half = weight.detach().to(torch.float16).t().contiguous()
        if not torch.isfinite(half).all():
            raise ValueError('FP16 weight overflow')
        handle = self.lib.mm_create(weight.shape[1], weight.shape[0], half.data_ptr())
        if not handle:
            raise self.error()
        self.handles.append(handle)
        self.create_seconds += time.perf_counter() - start
        return handle

    def run(self, handle, inputs, out_features):
        start = time.perf_counter()
        flat = inputs.reshape(-1, inputs.shape[-1]).to(torch.float16).contiguous()
        if not torch.isfinite(flat).all():
            raise ValueError('FP16 activation overflow')
        output = torch.empty((flat.shape[0], out_features), dtype=torch.float32)
        if self.lib.mm_run(handle, flat.shape[0], flat.data_ptr(), output.data_ptr()):
            raise self.error()
        self.calls += 1
        self.seconds += time.perf_counter() - start
        return output.reshape(*inputs.shape[:-1], out_features)

    def close(self):
        for handle in self.handles:
            self.lib.mm_destroy(handle)
        self.handles.clear()


def install(model, engine, selection='all'):
    """Replace only CPU BF16 text Linear.forward, after existing CPU optimization.

    Unsupported large requests use the saved CPU forward. No runtime error is
    silently hidden. Handles/weights live for the whole benchmark, not each call.
    """
    chosen = []
    for name, module in model.backbone.named_modules():
        if 'language_model.layers.' not in name or not isinstance(module, torch.nn.Linear):
            continue
        if selection == 'mlp' and '.mlp.' not in name:
            continue
        k, n = module.in_features, module.out_features
        if k > 10240 or k % 16 or n % 8 or k * n < 262144:
            continue
        module._rknn_cpu_forward = module.forward
        module._rknn_handle = None
        def forward(self, inputs):
            rows = inputs.numel() // inputs.shape[-1]
            if inputs.device.type != 'cpu' or inputs.dtype != torch.bfloat16 or rows > 512:
                engine.cpu_fallbacks += 1
                return self._rknn_cpu_forward(inputs)
            if self._rknn_handle is None:
                self._rknn_handle = engine.create(self.weight)
            output = engine.run(self._rknn_handle, inputs, self.out_features)
            if self.bias is not None:
                output += self.bias.float()
            return output.to(inputs.dtype)
        module.forward = MethodType(forward, module)
        chosen.append({'name': name, 'K': k, 'N': n})
    return chosen


def uninstall(model):
    # Restore CPU forwards before freeing NPU buffers, so retained model objects
    # cannot reference released native handles.
    for module in model.backbone.modules():
        if hasattr(module, '_rknn_cpu_forward'):
            module.forward = module._rknn_cpu_forward
            del module._rknn_cpu_forward
            del module._rknn_handle

#!/usr/bin/env python3
"""Serve pinned d3 with faster ARM CPU Linear/Conv kernels.

Keep BF16 checkpoint tensors and activation boundaries. Compute Linear and
Conv operations in FP32 using cached FP32 weights, then round outputs to the
original activation dtype. Probability parity must be measured before use.
"""
import argparse
import os
import sys
from pathlib import Path
from types import MethodType

import torch
import torch.nn.functional as functional


def linear_forward(self, inputs):
    return functional.linear(inputs.float(), self._cpu_fp32_weight,
                             self._cpu_fp32_bias).to(inputs.dtype)


def conv_forward(self, inputs):
    return self._conv_forward(inputs.float(), self._cpu_fp32_weight,
                             self._cpu_fp32_bias).to(inputs.dtype)


def optimize(model):
    count = 0
    cache_bytes = 0
    for module in model.backbone.modules():
        if not isinstance(module, (torch.nn.Linear, torch.nn.Conv1d,
                                   torch.nn.Conv2d, torch.nn.Conv3d)):
            continue
        if module.weight.device.type != "cpu" or module.weight.dtype != torch.bfloat16:
            continue
        weight = module.weight.detach().float()
        bias = None if module.bias is None else module.bias.detach().float()
        module.register_buffer("_cpu_fp32_weight", weight, persistent=False)
        module.register_buffer("_cpu_fp32_bias", bias, persistent=False)
        function = linear_forward if isinstance(module, torch.nn.Linear) else conv_forward
        module.forward = MethodType(function, module)
        count += 1
        cache_bytes += weight.numel() * weight.element_size()
        if bias is not None:
            cache_bytes += bias.numel() * bias.element_size()
    return count, cache_bytes


def main():

    parser.add_argument("--model", required=True, type=Path)
    parser.add_argument("--host", default="0.0.0.0")

        server.serve_forever()
    finally:
        server.server_close()


if __name__ == "__main__":
    main()
